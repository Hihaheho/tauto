// The data of the library prelude (`std/prelude/std.taut`) that proofs see
// too, as the kernel reads it: each as an inductive family.
pub(open) inductive Option(A: Type): Type {
  None: Option(A)
  Some(x0: A): Option(A)
}

pub(open) inductive List(A: Type): Type {
  Nil: List(A)
  Cons(x0: A, x1: List(A)): List(A)
}

pub(open) inductive Pair(A: Type, B: Type): Type {
  Mk(fst: A, snd: B): Pair(A, B)
}

pub(open) inductive Ordering: Type {
  Less: Ordering
  Equal: Ordering
  Greater: Ordering
}

pub(open) inductive Result(T: Type, E: Type): Type {
  Ok(x0: T): Result(T, E)
  Err(x0: E): Result(T, E)
}
